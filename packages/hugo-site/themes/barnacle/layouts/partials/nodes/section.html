{{/* nodes/section.html — renders a PageNode of type "section". Emits a semantic
element. styleRefs are resolved to CSS class names from graph.styles and applied to the class attribute. */}} {{- $node := .node -}} {{- $nodeMap := .nodeMap -}} {{- $graph := .graph -}} {{- /* Resolve styleRefs → space-separated class list. */}} {{- $classes := slice -}} {{- range $node.styleRefs -}} {{- $style := index $graph.styles . -}} {{- if $style -}} {{- $classes = append $style.className $classes -}} {{- end -}} {{- end -}} {{- $classAttr := delimit $classes " " -}}
{{- range $node.children -}} {{- $child := index $nodeMap . -}} {{- if $child -}} {{- partial "pagegraph-node.html" (dict "node" $child "nodeMap" $nodeMap "graph" $graph) -}} {{- end -}} {{- end -}}
